================================================
Best Seller
================================================

 VAR TopProduct = TOPN(
                        1,
                        ADDCOLUMNS(
                            VALUES(products[product_name]), 
                            "TotalQty", [Total Items Sold]
                        )
                        ,
                        [TotalQty],
                        DESC
 )
RETURN
    MAXX(TopProduct, products[product_name])

================================================
Top Category
================================================

VAR TopCat = TOPN(
                1,
                 SUMMARIZE(
                    order_details, 
                    products[category],
                    "Category Qty", SUM(order_details[quantity])
                )
)
RETURN 
    MAXX(TopCat, products[category])

================================================
Total Product
================================================

VAR CurrentDate =
    MAX('date'[Date])

RETURN
CALCULATE(
    DISTINCTCOUNT(products[product_id]),
    FILTER(
        products,
        products[launch_date] <= CurrentDate
))

================================================
Total Purchased Product
================================================

DISTINCTCOUNT(order_details[product_id])



================================================
Avg per Product

AVERAGEX(

)

Max Revenue per Product

Min Revenue per Product

Product Best Revenue

VAR BestProduct = TOPN(
    1,
    ADDCOLUMNS(
        VALUES(products[product_name]),
        "TotalRevenue", [Total Revenue]
    ),
    [TotalRevenue],
    DESC
    
)

RETURN

================================================
Product Bad Revenue
================================================

VAR BadProduct =
    TOPN(
        1,
        ADDCOLUMNS(
            VALUES(order_details[product_id]),
            "Revenue", [Total Revenue]
        ),
        [Revenue], ASC,
        order_details[product_id], ASC
    )
RETURN
MAXX(
    BadProduct,
    LOOKUPVALUE(
        products[product_name],
        products[product_id],
        order_details[product_id]
    )
)
