================================================
MoM Total Sales
================================================

VAR LatestMonth = CALCULATE([Total Revenue], DATEADD('date'[Date], -1, MONTH))
VAR diff = [Total Revenue] - LatestMonth
RETURN
    SWITCH(
        TRUE(),
         diff > 0, UNICHAR(9650) & " " & FORMAT(DIVIDE(diff, LatestMonth), "0.00%") & " Vs last Month",
         diff < 0, UNICHAR(9660) & " " & FORMAT(DIVIDE(diff, LatestMonth), "0.00%") & " Vs last Month",
        UNICHAR(61) & " " & FORMAT(DIVIDE(diff, LatestMonth), "0.00%"))

================================================
MoM Profit Margin
================================================

    VAR LatestMonth =
        CALCULATE(
            [Profit Margin],
            DATEADD('date'[Date], -1, MONTH)
        )

    VAR Growth =
        DIVIDE(
            [Profit Margin] - LatestMonth,
            LatestMonth
        )

    RETURN
    SWITCH(
        TRUE(),
        ISBLANK(LatestMonth), BLANK(),

        Growth > 0,
            UNICHAR(9650) & " " &
            FORMAT(Growth, "0.00%") &
            " vs Last Month",

        Growth < 0,
            UNICHAR(9660) & " " &
            FORMAT(ABS(Growth), "0.00%") &
            " vs Last Month",

        UNICHAR(61) & " 0.00% vs Last Month"
    )
